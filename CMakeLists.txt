cmake_minimum_required(VERSION 3.20)
project(book_replay VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(book_replay src/book.cpp)
target_include_directories(book_replay PUBLIC include)
if(MSVC)
  target_compile_options(book_replay PRIVATE /W4 /permissive-)
else()
  target_compile_options(book_replay PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_executable(book-replay src/main.cpp)
target_link_libraries(book-replay PRIVATE book_replay)
include(CTest)
if(BUILD_TESTING)
  add_executable(book-tests tests/book_tests.cpp)
  target_link_libraries(book-tests PRIVATE book_replay)
  add_test(NAME book-tests COMMAND book-tests)
endif()
